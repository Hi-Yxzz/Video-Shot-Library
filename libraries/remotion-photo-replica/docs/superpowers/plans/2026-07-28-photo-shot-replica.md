# Photo Shot Replica Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an editable Remotion Studio composition that turns the supplied portrait into a silent 7.67-second shot with subtle parallax and two replaceable looping element tracks.

**Architecture:** A typed Zod schema is the public editing interface. `PhotoReplica` owns the timeline, `PortraitScene` owns the background/soft-mask parallax, and reusable `LoopingTrack`/`ElementCard` components render data-driven foreground and background tracks. Pure loop math is isolated and unit-tested; representative frames and the final MP4 provide integration verification.

**Tech Stack:** React, TypeScript, Remotion, Zod, Vitest, FFmpeg/FFprobe

---

import {z} from 'zod';

const elementItemSchema = z.object({
  id: z.string().min(1),
  type: z.enum(['text', 'image', 'mixed']),
  label: z.string(),
  imageSrc: z.string().optional(),
  accentColor: z.string().regex(/^#[0-9a-fA-F]{6}$/),
});

const trackSchema = z.object({
  items: z.array(elementItemSchema),
  speed: z.number().min(0).max(10),
  direction: z.enum(['left', 'right']),
  gap: z.number().min(0).max(200),
  cardWidth: z.number().min(40).max(500),
  cardHeight: z.number().min(30).max(300),
  opacity: z.number().min(0).max(1),
  blur: z.number().min(0).max(20),
  borderRadius: z.number().min(0).max(100),
  shadow: z.number().min(0).max(1),
});

export const photoReplicaSchema = z.object({
  portraitSrc: z.string().min(1),
  durationInFrames: z.number().int().min(30).max(1800),
  backgroundScale: z.number().min(1).max(1.3),
  backgroundPositionX: z.number().min(0).max(100),
  backgroundPositionY: z.number().min(0).max(100),
  portraitScale: z.number().min(1).max(1.2),
  portraitOffsetX: z.number().min(-200).max(200),
  portraitOffsetY: z.number().min(-200).max(200),
  breathingAmount: z.number().min(0).max(0.05),
  floatingAmount: z.number().min(0).max(30),
  cameraPushAmount: z.number().min(0).max(0.1),
  backgroundBrightness: z.number().min(0.5).max(1.5),
  backgroundContrast: z.number().min(0.5).max(1.5),
  vignetteAmount: z.number().min(0).max(1),
  topTrack: trackSchema,
  bottomTrack: trackSchema,
});

export type PhotoReplicaProps = z.infer<typeof photoReplicaSchema>;
export type ElementItem = PhotoReplicaProps['topTrack']['items'][number];
export type TrackConfig = PhotoReplicaProps['topTrack'];

Create `src/lib/looping.ts`:

```ts
type LoopOffsetInput = {
  frame: number;
  speed: number;
  unitWidth: number;
  direction: 'left' | 'right';
};

export const getLoopOffset = ({frame, speed, unitWidth, direction}: LoopOffsetInput) => {
  if (speed === 0 || unitWidth <= 0) return 0;
  const progress = (frame * speed) % unitWidth;
  return direction === 'left' ? -progress : -unitWidth + progress;
};

export const getRepeatCount = (viewportWidth: number, unitWidth: number) => {
  if (unitWidth <= 0) return 0;
  return Math.ceil(viewportWidth / unitWidth) + 2;
};
```

import {useState, type CSSProperties} from 'react';
import {Img, staticFile} from 'remotion';
import type {ElementItem} from '../config';

export const ElementCard = ({
  item,
  width,
  height,
  borderRadius,
  shadow,
}: {
  item: ElementItem;
  width: number;
  height: number;
  borderRadius: number;
  shadow: number;
}) => {
  const [imageFailed, setImageFailed] = useState(false);
  const showImage = (item.type === 'image' || item.type === 'mixed') && item.imageSrc && !imageFailed;
  const showLabel = item.type !== 'image' || !showImage;
  const style: CSSProperties = {
    width,
    height,
    flex: '0 0 auto',
    borderRadius,
    boxSizing: 'border-box',
    display: 'flex',
    alignItems: 'center',
    justifyContent: 'center',
    gap: 10,
    padding: '10px 14px',
    overflow: 'hidden',
    background: 'linear-gradient(145deg, rgba(255,255,255,.96), rgba(236,239,244,.88))',
    border: `1px solid ${item.accentColor}55`,
    boxShadow: `0 12px 30px rgba(0,0,0,${shadow})`,
    color: '#151922',
    fontFamily: 'Arial, sans-serif',
    fontSize: 19,
    fontWeight: 700,
  };

  return (
    <div style={style}>
      {showImage ? (
        <Img
          src={staticFile(item.imageSrc!)}
          onError={() => setImageFailed(true)}
          style={{maxWidth: 44, maxHeight: 44, objectFit: 'contain'}}

- [ ] **Step 4: Implement the looping track**

Create `src/components/LoopingTrack.tsx`:

```tsx
import {useCurrentFrame, useVideoConfig} from 'remotion';
import type {TrackConfig} from '../config';
import {getLoopOffset, getRepeatCount} from '../lib/looping';
import {ElementCard} from './ElementCard';

export const LoopingTrack = ({config, name}: {config: TrackConfig; name: string}) => {
  const frame = useCurrentFrame();
  const {width} = useVideoConfig();
  if (config.items.length === 0) return null;

  const unitWidth = config.items.length * (config.cardWidth + config.gap);
  const repeatCount = getRepeatCount(width, unitWidth);
  const offset = getLoopOffset({
    frame,
    speed: config.speed,
    unitWidth,
    direction: config.direction,
  });

  return (
    <div aria-label={name} style={{width: '100%', overflow: 'hidden', opacity: config.opacity, filter: `blur(${config.blur}px)`}}>
      <div style={{display: 'flex', gap: config.gap, width: 'max-content', translate: `${offset}px 0`}}>
        {Array.from({length: repeatCount}).flatMap((_, repeatIndex) =>
          config.items.map((item) => (
            <ElementCard
              key={`${repeatIndex}-${item.id}`}
              item={item}
              width={config.cardWidth}
              height={config.cardHeight}
              borderRadius={config.borderRadius}
              shadow={config.shadow}
            />
          )),
        )}
      </div>
    </div>
  );
};
```

```tsx
import {AbsoluteFill, Img, interpolate, staticFile, useCurrentFrame, useVideoConfig} from 'remotion';
import type {PhotoReplicaProps} from '../config';

type PortraitProps = Pick<
  PhotoReplicaProps,
  | 'portraitSrc'
  | 'backgroundScale'
  | 'backgroundPositionX'
  | 'backgroundPositionY'
  | 'portraitScale'
  | 'portraitOffsetX'
  | 'portraitOffsetY'
  | 'breathingAmount'
  | 'floatingAmount'
  | 'cameraPushAmount'
  | 'backgroundBrightness'
  | 'backgroundContrast'
  | 'vignetteAmount'
>;

export const PortraitScene = (props: PortraitProps) => {
  const frame = useCurrentFrame();
  const {durationInFrames} = useVideoConfig();
  const phase = (frame / 90) * Math.PI * 2;
  const cameraPush = interpolate(frame, [0, durationInFrames - 1], [0, props.cameraPushAmount], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });
  const photoStyle = {
    width: '100%',
    height: '100%',
    objectFit: 'cover' as const,
    objectPosition: `${props.backgroundPositionX}% ${props.backgroundPositionY}%`,
  };

  return (
    <AbsoluteFill style={{overflow: 'hidden', backgroundColor: '#bec3c6'}}>
      <AbsoluteFill style={{scale: props.backgroundScale + cameraPush}}>
        <Img
          src={staticFile(props.portraitSrc)}
          style={{...photoStyle, filter: `brightness(${props.backgroundBrightness}) contrast(${props.backgroundContrast})`}}
        />
      </AbsoluteFill>
      <AbsoluteFill
        style={{
          translate: `${props.portraitOffsetX}px ${props.portraitOffsetY + Math.sin(phase) * props.floatingAmount}px`,
          scale: props.portraitScale + cameraPush * 1.15 + Math.sin(phase) * props.breathingAmount,
          WebkitMaskImage: 'radial-gradient(ellipse 31% 69% at 50% 53%, #000 68%, transparent 100%)',
          maskImage: 'radial-gradient(ellipse 31% 69% at 50% 53%, #000 68%, transparent 100%)',
        }}
      >
        <Img src={staticFile(props.portraitSrc)} style={photoStyle} />
      </AbsoluteFill>
      <AbsoluteFill
        style={{
          pointerEvents: 'none',
          background: `radial-gradient(ellipse at center, transparent 48%, rgba(0,0,0,${props.vignetteAmount}) 100%)`,
        }}
      />
    </AbsoluteFill>
  );
};
```

import {AbsoluteFill, Easing, interpolate, useCurrentFrame} from 'remotion';
import type {PhotoReplicaProps} from './config';
import {LoopingTrack} from './components/LoopingTrack';
import {PortraitScene} from './components/PortraitScene';

export const PhotoReplica = (props: PhotoReplicaProps) => {
  const frame = useCurrentFrame();
  const entrance = interpolate(frame, [0, 24], [0, 1], {
    easing: Easing.bezier(0.16, 1, 0.3, 1),
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });

  return (
    <AbsoluteFill style={{backgroundColor: '#bfc4c7', opacity: entrance}}>
      <PortraitScene {...props} />
      <AbsoluteFill style={{justifyContent: 'flex-start', paddingTop: 54, zIndex: 1}}>
        <LoopingTrack config={props.topTrack} name="Background element track" />
      </AbsoluteFill>
      <AbsoluteFill
        style={{
          zIndex: 2,
          pointerEvents: 'none',
          WebkitMaskImage: 'radial-gradient(ellipse 31% 69% at 50% 53%, #000 68%, transparent 100%)',
          maskImage: 'radial-gradient(ellipse 31% 69% at 50% 53%, #000 68%, transparent 100%)',
        }}
      >
        <PortraitScene {...props} />
      </AbsoluteFill>
      <AbsoluteFill style={{justifyContent: 'flex-end', paddingBottom: 42, zIndex: 3}}>
        <LoopingTrack config={props.bottomTrack} name="Foreground element track" />
      </AbsoluteFill>
    </AbsoluteFill>
  );
};
